前端性能优化：防抖 & 节流

防抖函数 (debounce) 代码：
function debounce(func, delay) {
    let timer; // 定义一个定时器变量，用于存储setTimeout的返回值
    return function() {
        clearTimeout(timer); // 每次触发时，清除之前的定时器，取消之前等待执行的函数
        timer = setTimeout(() => { // 重新设置一个定时器，delay毫秒后执行传入的函数
            func.apply(this, arguments); // 修正this指向并传递所有参数
        }, delay);
    }
}
节流函数 (throttle) 代码：
function throttle(func, delay) {
    let lastTime = 0; // 记录上一次函数执行的时间戳
    return function() {
        const now = Date.now(); // 获取当前时间戳
        // 判断当前时间与上次执行时间的差值是否大于等于指定的延迟时间
        if (now - lastTime >= delay) {
            func.apply(this, arguments); // 执行原函数并传递参数
            lastTime = now; // 更新上一次执行的时间戳为当前时间
        }
    }
}
